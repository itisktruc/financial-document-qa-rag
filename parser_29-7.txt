_VN_DIACRITICS_RE = re.compile(
    "[ăâêôơưđáàảãạấầẩẫậắằẳẵặéèẻẽẹếềểễệíìỉĩịóòỏõọốồổỗộớờởỡợúùủũụứừửữựýỳỷỹỵ]",
    re.IGNORECASE,
)
_SUSPICIOUS_TOKEN_RE = re.compile(rf"^[A-Za-z]{{{SUSPICIOUS_TOKEN_MIN_LEN},}}$")
_SINGLE_LETTER_RE = re.compile(r"^[A-Za-z]$")


def _bbox_too_small(bbox: list) -> bool:
    """True nếu bbox quá nhỏ để là 1 vùng chữ thật (nhiều khả năng là nhiễu)."""
    w = bbox[2] - bbox[0]
    h = bbox[3] - bbox[1]
    if w <= 0 or h <= 0:
        return True
    if w < MIN_BOX_WIDTH_PX or h < MIN_BOX_HEIGHT_PX:
        return True
    if w * h < MIN_BOX_AREA_PX:
        return True
    return False


def _looks_like_stray_char(text: str) -> bool:
    """True nếu kết quả nhận diện CHỈ là 1 ký tự Latin đơn lẻ - lớp phòng thủ
    thứ 2 (bổ sung cho _bbox_too_small) cho trường hợp bbox không quá nhỏ
    nhưng vẫn không phải vùng chữ thật."""
    return bool(_SINGLE_LETTER_RE.match(text.strip()))


def _is_suspicious_hallucination(text: str) -> bool:
    """True nếu trông giống 1 chuỗi giả-tiếng-Anh do vietocr "bịa" ra: toàn chữ
    Latin liên tục, đủ dài, KHÔNG có dấu tiếng Việt - bất thường với tài liệu
    tiếng Việt (số liệu/đơn vị thật thường là số, ký hiệu, hoặc có dấu)."""
    t = text.strip()
    if not _SUSPICIOUS_TOKEN_RE.match(t):
        return False
    return not _VN_DIACRITICS_RE.search(t)


def _build_vietocr_predictor() -> Predictor:
    config = Cfg.load_config_from_name(VIETOCR_MODEL_NAME)
    config["device"] = VIETOCR_DEVICE
    if VIETOCR_WEIGHTS:
        config["weights"] = VIETOCR_WEIGHTS
    return Predictor(config)


_vietocr_predictor = _build_vietocr_predictor()

"""
ppv3_worker_class_fix.py

Đây là bản thay thế cho class PPV3Worker trong parser.py - dán đè lên class
PPV3Worker cũ (giữ nguyên phần import ở đầu file parser.py, chỉ thay class
này + phần khởi tạo _ppv3_worker/atexit ở dưới nó).

BUG ĐÃ SỬA: bản cũ chỉ đọc stdout của subprocess, không bao giờ đọc stderr.
PPStructureV3 khi load model in RẤT NHIỀU ra stderr (tqdm progress, cảnh báo
ccache, thông báo tải model...). Pipe OS có buffer giới hạn (~64KB) - khi đầy,
subprocess con bị BLOCK ngay tại lệnh in tiếp theo, không bao giờ chạy tới
dòng in {"ready": true} ra stdout -> process cha (đang gọi readline() chờ
stdout) bị treo VÔ THỜI HẠN, không báo lỗi, không timeout. Đây chính là
nguyên nhân "kẹt hơn 4 tiếng" mà không có bất kỳ traceback nào.

CÁCH SỬA: dùng 2 thread nền, mỗi thread liên tục đọc 1 trong 2 pipe (stdout,
stderr) và đẩy dòng đọc được vào hàng đợi (queue.Queue) - không bao giờ để
pipe nào bị đầy vì luôn có người đọc. Thread chính chỉ lấy dòng cần thiết
(JSON kết quả) từ queue của stdout, có timeout rõ ràng - nếu quá thời gian
mà không có kết quả, báo lỗi ngay thay vì treo im lặng.
"""
# Timeout khởi động worker (load model PPStructureV3 lần đầu có thể mất vài
# phút nếu phải tải model qua mạng lần đầu - để dư thời gian nhưng vẫn có
# giới hạn, không treo vô thời hạn như bản cũ).
WORKER_STARTUP_TIMEOUT = int(os.getenv("PPV3_WORKER_STARTUP_TIMEOUT", 900))
# Timeout xử lý 1 trang - nếu 1 trang OCR quá lâu (nghi hỏng/kẹt), báo lỗi
# thay vì treo cả pipeline.
WORKER_PAGE_TIMEOUT = int(os.getenv("PPV3_WORKER_PAGE_TIMEOUT", 180))


def _pipe_reader_thread(pipe, out_queue: "queue.Queue[str]", tag: str):
    """
    Đọc liên tục 1 pipe (stdout hoặc stderr) theo từng dòng, đẩy vào queue.
    Chạy trong thread riêng - mục đích DUY NHẤT là đảm bảo pipe không bao
    giờ bị đầy (nguyên nhân gây deadlock ở bản cũ).
    """
    try:
        for line in iter(pipe.readline, ""):
            if tag == "stderr":
                # In log của worker ra kèm nhãn để dễ debug, KHÔNG đẩy vào
                # queue kết quả (chỉ stdout mới chứa JSON kết quả thật).
                print(f"[ppv3_worker:{tag}] {line.rstrip()}", flush=True)
            else:
                out_queue.put(line)
    except Exception:
        pass
    finally:
        # BUG ĐÃ SỬA: trước đây CẢ 2 luồng (stdout lẫn stderr) đều đẩy None
        # vào CHUNG 1 queue kết quả khi pipe của chúng đóng. Vì 2 pipe của
        # cùng 1 process thường đóng gần như đồng thời lúc worker thoát, kết
        # quả là hàng đợi nhận thêm 1 mục "None" thừa không tương ứng với bất
        # kỳ request nào -> đủ để làm lệch nhịp (mis-align) các lần đọc tiếp
        # theo, khiến process cha đọc nhầm kết quả của trang này cho trang
        # khác. CHỈ luồng đọc STDOUT (nơi thực sự mang JSON kết quả) mới được
        # phép báo hiệu "worker đã thoát" vào hàng đợi kết quả.
        if tag != "stderr":
            out_queue.put(None)


class PPV3Worker:
    """Quản lý subprocess dài hạn chạy PPStructureV3, đọc pipe an toàn qua thread+queue."""

    def __init__(self):
        self._proc: "subprocess.Popen | None" = None
        self._lock = threading.Lock()
        self._stdout_queue: "queue.Queue[str]" = queue.Queue()
        self._stderr_thread: "threading.Thread | None" = None
        self._stdout_thread: "threading.Thread | None" = None

    def _ensure_started(self):
        if self._proc is not None and self._proc.poll() is None:
            return

        ppv3_worker_script = os.path.join(
            os.path.dirname(os.path.abspath(__file__)), "ppv3_worker.py"
        )

        self._stdout_queue = queue.Queue()
        self._proc = subprocess.Popen(
            [PADDLE_VENV_PYTHON, ppv3_worker_script],
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            bufsize=1,
            env={**os.environ, "OCR_DEVICE": OCR_DEVICE, "OCR_CPU_THREADS": str(CPU_THREADS)},
        )

        # 2 thread nền đọc song song stdout/stderr - KHÔNG BAO GIỜ để 1 trong
        # 2 pipe bị đầy (đây là fix chính cho bug deadlock).
        self._stdout_thread = threading.Thread(
            target=_pipe_reader_thread,
            args=(self._proc.stdout, self._stdout_queue, "stdout"),
            daemon=True,
        )
        self._stderr_thread = threading.Thread(
            target=_pipe_reader_thread,
            args=(self._proc.stderr, self._stdout_queue, "stderr"),
            daemon=True,
        )
        self._stdout_thread.start()
        self._stderr_thread.start()

        # Chờ dòng "ready" thật, nhưng CHỊU ĐƯỢC dòng rác không phải JSON lọt
        # vào giữa chừng (vd log/banner của thư viện con vô tình in ra stdout
        # trước khi ppv3_worker.py kịp redirect) - bỏ qua và tiếp tục chờ,
        # dùng CHUNG 1 deadline tổng (không reset lại đồng hồ mỗi dòng rác,
        # tránh treo vô hạn nếu worker liên tục in rác thay vì "ready" thật).
        deadline = time.monotonic() + WORKER_STARTUP_TIMEOUT
        while True:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                self._kill_proc()
                raise RuntimeError(
                    f"Worker PPStructureV3 không khởi động xong sau "
                    f"{WORKER_STARTUP_TIMEOUT}s - xem log [ppv3_worker:stderr] ở "
                    f"trên để biết đang kẹt ở bước nào (thường là tải model qua "
                    f"mạng nếu chạy lần đầu)."
                )
            try:
                ready_line = self._stdout_queue.get(timeout=remaining)
            except queue.Empty:
                self._kill_proc()
                raise RuntimeError(
                    f"Worker PPStructureV3 không khởi động xong sau "
                    f"{WORKER_STARTUP_TIMEOUT}s - xem log [ppv3_worker:stderr] ở "
                    f"trên để biết đang kẹt ở bước nào (thường là tải model qua "
                    f"mạng nếu chạy lần đầu)."
                )

            if ready_line is None:
                raise RuntimeError("Worker PPStructureV3 thoát ngay sau khi khởi động, xem log stderr ở trên.")

            try:
                ready = json.loads(ready_line)
            except json.JSONDecodeError:
                # Dòng rác - KHÔNG coi là lỗi khởi động, tiếp tục chờ.
                print(f"[ppv3_worker] Bỏ qua dòng không phải JSON lúc chờ ready: {ready_line!r}", flush=True)
                continue

            if not ready.get("ready"):
                self._kill_proc()
                raise RuntimeError(f"Worker PPStructureV3 không khởi động được: {ready_line!r}")
            break

    def get_structure(self, image: np.ndarray) -> dict:
        with self._lock:
            self._ensure_started()

            import tempfile
            import uuid

            npy_path = os.path.join(tempfile.gettempdir(), f"{uuid.uuid4().hex}.npy")
            np.save(npy_path, image)
            try:
                self._proc.stdin.write(json.dumps({"npy_path": npy_path}) + "\n")
                self._proc.stdin.flush()

                try:
                    line = self._stdout_queue.get(timeout=WORKER_PAGE_TIMEOUT)
                except queue.Empty:
                    self._kill_proc()
                    return {"error": f"Worker không phản hồi sau {WORKER_PAGE_TIMEOUT}s (nghi bị treo/OOM)."}

                if line is None:
                    self._proc = None
                    return {"error": "Worker PPStructureV3 đã thoát bất ngờ giữa chừng."}

                return json.loads(line)
            finally:
                try:
                    os.remove(npy_path)
                except OSError:
                    pass

    def _kill_proc(self):
        if self._proc is not None:
            try:
                self._proc.kill()
            except Exception:
                pass
            self._proc = None

    def shutdown(self):
        if self._proc is not None and self._proc.poll() is None:
            try:
                self._proc.stdin.write("__QUIT__\n")
                self._proc.stdin.flush()
                self._proc.wait(timeout=5)
            except Exception:
                self._kill_proc()


_ppv3_worker = PPV3Worker()
atexit.register(_ppv3_worker.shutdown)

def has_text_layer(page, min_chars: int = 20) -> bool:
    """Kiểm tra text layer của MỘT trang (thay vì cả tài liệu)."""
    textpage = page.get_textpage()
    text = textpage.get_text_range().strip()
    return len(text) >= min_chars, text


def _render_page_array(page, dpi: int) -> np.ndarray:
    """Render trang PDF thành numpy array trong RAM, không ghi ra đĩa."""
    scale = dpi / 72
    bitmap = page.render(scale=scale)
    pil_image = bitmap.to_pil().convert("RGB")
    return np.array(pil_image)


def _recognize_crop(image: np.ndarray, bbox: list) -> tuple:
    """Cắt 1 vùng bbox khỏi ảnh trang (kèm đệm nhỏ) và nhận diện chữ bằng vietocr."""
    # FIX: loại bbox nhiễu (viền trang/vạch lề/dấu photo...) TRƯỚC KHI cắt ảnh
    # và gọi vietocr - đây là nguồn gốc chính của các dòng nhiễu "E"/"L"/"F".
    if _bbox_too_small(bbox):
        return "", 0.0

    h, w = image.shape[:2]
    x0 = max(int(bbox[0]) - CROP_PADDING_PX, 0)
    y0 = max(int(bbox[1]) - CROP_PADDING_PX, 0)
    x1 = min(int(bbox[2]) + CROP_PADDING_PX, w)
    y1 = min(int(bbox[3]) + CROP_PADDING_PX, h)
    if x1 <= x0 or y1 <= y0:
        return "", 0.0
    crop = image[y0:y1, x0:x1]
    try:
        text, prob = _vietocr_predictor.predict(Image.fromarray(crop), return_prob=True)
    except Exception:
        return "", 0.0

    prob = float(prob)
    if not math.isfinite(prob):
        # FIX: prob NaN/inf (quan sát được ở 1 số crop suy biến) trước đây làm
        # np.mean() của cả trang ra NaN -> so sánh "NaN < 0.85" luôn False trong
        # Python -> trang lỗi nặng nhất lại lọt qua danh sách cần review. Coi
        # prob không hữu hạn là "không tin cậy" (0.0), không phải bỏ qua.
        prob = 0.0
    return text, prob

    """

    văn tự nhiên - quan sát thực tế cho thấy nó hay "collapse" về vài chuỗi
    giả-tiếng-Anh vô nghĩa (Contraction, States...), đôi khi với prob báo cáo
    khá cao (confidently wrong). Vì vậy với TỪNG ô: nếu prob thấp HOẶC text
    trông giống hallucination (chuỗi Latin dài, không dấu, bất thường với tài
    liệu tiếng Việt) -> KHÔNG ghi đè, giữ lại text gốc paddle đã nhận diện cho
    ô đó (ít nhất còn dữ liệu tham khảo, không tệ hơn trước khi có vietocr) -
    đồng thời ghi nhận ô này vào `flagged` để người dùng biết chính xác ô nào
    cần đối chiếu lại với ảnh PDF gốc, thay vì phải dò cả bảng.

    GIẢ ĐỊNH CẦN VERIFY TRÊN TÀI LIỆU THẬT (xem ghi chú ppv3_worker.py): thứ tự
    bbox trong `cells` khớp với thứ tự thẻ <td>/<th> xuất hiện trong pred_html.
    Nếu số lượng lệch nhau -> KHÔNG cố gán bừa (tránh gán nhầm số liệu tài
    chính vào sai ô), giữ nguyên pred_html gốc (chữ do paddle nhận diện) để ít
    nhất vẫn có dữ liệu tham khảo, đồng thời báo hiệu qua confidence rỗng để
    lọt vào diện cần review thủ công.
    """
    if not pred_html or not cells:
        return pred_html, [], []

    soup = BeautifulSoup(pred_html, "html.parser")
    cell_tags = soup.find_all(["td", "th"])

    if len(cell_tags) != len(cells):
        return pred_html, [], []

    confidences = []
    flagged = []
    for tag, cell in zip(cell_tags, cells):
        bbox = cell.get("bbox", [0, 0, 0, 0])
        if bbox == [0, 0, 0, 0]:
            continue

        paddle_text = tag.get_text(strip=True)
        vietocr_text, prob = _recognize_crop(image, bbox)

        use_fallback = False
        if not vietocr_text:
            # vietocr không nhận diện được gì (bbox nhiễu/quá nhỏ/lỗi) - nếu
            # paddle vốn ĐÃ thấy có chữ ở đây thì đây là 1 lần thất bại thật
            # sự, cần fallback + flag; nếu paddle cũng thấy trống thì đây chỉ
            # là 1 ô trống bình thường, không có gì phải báo động.
            if paddle_text:
                use_fallback = True
        elif prob < CONF_FALLBACK_THRESHOLD or _is_suspicious_hallucination(vietocr_text):
            use_fallback = True

        if use_fallback:
            final_text = paddle_text
            flagged.append(
                {
                    "paddle_text": paddle_text,
                    "vietocr_text": vietocr_text,
                    "prob": round(prob, 4),
                }
            )
            confidences.append(prob)
        else:
            final_text = vietocr_text
            confidences.append(prob)

        tag.clear()
        tag.append(final_text)

    return str(soup), confidences, flagged


def _assemble_page(image: np.ndarray, structure: dict) -> dict:
    """
    Cắt ảnh theo bbox mà PP-StructureV3 phát hiện (subprocess) và nhận diện
    chữ bằng vietocr (venv chính). Trả về markdown của TRANG + độ tin cậy
    trung bình (trung bình prob của vietocr trên toàn trang) + danh sách ô
    bảng bị fallback/cần review thủ công (nếu có).
    """
    all_confidences = []

    # ---- đoạn văn ngoài bảng: sắp theo thứ tự đọc (trên->dưới, trái->phải) ----
    # ĐƠN GIẢN HOÁ: gộp các box có cùng "hàng" (lệch y trong khoảng 10px) rồi
    # sắp theo x - đủ tốt cho layout 1 cột, CHƯA xử lý layout nhiều cột phức
    # tạp (nếu tài liệu có nhiều cột, cần logic phức tạp hơn dựa vào x-range).
    text_boxes = sorted(
        (tb["bbox"] for tb in structure.get("text_boxes", [])),
        key=lambda b: (round(b[1] / 10), b[0]),
    )
    paragraph_lines = []
    for bbox in text_boxes:
        text, prob = _recognize_crop(image, bbox)
        if not text:
            continue
        if _looks_like_stray_char(text):
            # FIX: lớp phòng thủ thứ 2 (sau _bbox_too_small ở _recognize_crop) -
            # 1 dòng CHỈ gồm 1 ký tự Latin đơn lẻ hầu như luôn là nhiễu (viền
            # trang/watermark/dấu photo), không phải nội dung thật -> bỏ qua,
            # không đưa vào markdown, không tính vào confidence trung bình.
            continue
        paragraph_lines.append(text)
        all_confidences.append(prob)
    paragraph_text = "\n".join(paragraph_lines)

    # ---- bảng: giữ khung HTML của paddle, thay chữ bằng vietocr (có fallback) ----
    table_htmls = []
    flagged_cells = []
    for tbl in structure.get("tables", []):
        html, confs, flagged = _fill_table_html(tbl.get("pred_html", ""), tbl.get("cells", []), image)
        table_htmls.append(html)
        all_confidences.extend(confs)
        flagged_cells.extend(flagged)

    markdown = "\n\n".join([p for p in [paragraph_text] if p] + table_htmls)

    # FIX: lọc bỏ giá trị prob không hữu hạn TRƯỚC KHI tính trung bình - dù
    # _recognize_crop đã sanitize tại nguồn, giữ thêm lớp phòng thủ ở đây để
    # np.mean() không bao giờ có thể ra NaN (tránh lặp lại bug trang 26 trước
    # đó: NaN < 0.85 luôn False -> trang lỗi nặng nhất lại không được review).
    finite_confidences = [c for c in all_confidences if math.isfinite(c)]
    avg_conf = float(np.mean(finite_confidences)) if finite_confidences else 0.0

    return {"markdown": markdown, "confidence": avg_conf, "flagged_cells": flagged_cells}


def _structure_page(image: np.ndarray) -> dict:
    """
    Lấy layout/bbox từ PP-StructureV3 (subprocess) rồi nhận diện chữ bằng
    vietocr (_assemble_page). Bọc try/except vì lỗi IPC (worker crash do OOM
    GPU, mất kết nối...) không được để crash toàn bộ tài liệu.
    """
    try:
        structure = _ppv3_worker.get_structure(image)
    except Exception as e:
        return {"markdown": "", "confidence": 0.0, "flagged_cells": [], "error": str(e)}

    if "error" in structure:
        return {"markdown": "", "confidence": 0.0, "flagged_cells": [], "error": structure["error"]}

    return _assemble_page(image, structure)


def parse_pdf(pdf_path: str) -> dict:
    """
    Entry point: parse PDF -> structured output.
    Trang có text layer -> lấy text native (nhanh, chính xác tuyệt đối).
    Trang không có text layer -> PP-StructureV3 phát hiện layout/bảng (subprocess,
    venv riêng) + vietocr nhận diện chữ trên từng vùng đã cắt (venv chính).
    """
    pdf = pdfium.PdfDocument(pdf_path)
    num_pages = len(pdf)

    # Nếu đa số trang có text layer thật -> vẫn ưu tiên docling cho toàn văn bản
    # (giữ nguyên bảng biểu/markdown structure tốt hơn OCR thủ công).
    sample = min(5, num_pages)
    native_count = sum(has_text_layer(pdf[i])[0] for i in range(sample))
    if native_count >= (sample // 2 + 1):
        pdf.close()
        result = _docling_converter.convert(pdf_path)
        return {
            "source": "docling_native",
            "markdown": result.document.export_to_markdown(),
            "tables": result.document.tables,
            "pages": result.document.num_pages(),
        }

    # Nhánh scan / mixed: xử lý per-page, chỉ OCR trang thật sự cần
    pages_markdown = {}
    pages_confidence = {}   # 1.0 = native text (chắc chắn đúng); < 1.0 = qua OCR
    pages_error = {}        # lưu lại lỗi thật (nếu có) thay vì nuốt mất -> dễ debug
    pages_flagged_cells = {}  # ô bảng bị fallback (vietocr nghi ảo giác) - cần review

    for i in range(num_pages):
        page = pdf[i]
        has_native, native_text = has_text_layer(page)
        if has_native:
            pages_markdown[i + 1] = native_text
            pages_confidence[i + 1] = 1.0
        else:
            img = _render_page_array(page, OCR_DPI)
            structured = _structure_page(img)
            pages_markdown[i + 1] = structured["markdown"]
            pages_confidence[i + 1] = round(structured["confidence"], 4)
            if structured.get("flagged_cells"):
                pages_flagged_cells[i + 1] = structured["flagged_cells"]
            if structured.get("error"):
                pages_error[i + 1] = structured["error"]
                # In NGAY ra console lúc chạy -- đừng đợi tới lúc mở file JSON mới
                # biết trang nào lỗi và lỗi gì, nhất là khi mọi trang lỗi giống nhau.
                print(f"[!] Trang {i + 1} lỗi: {structured['error']}", flush=True)

    pdf.close()

    # FIX: dùng "not (c >= 0.85)" thay vì "c < 0.85" - an toàn với NaN (so sánh
    # trực tiếp với NaN luôn ra False trong Python, khiến trang lỗi nặng nhất
    # lại lọt qua danh sách cần review - xem ghi chú ở _recognize_crop/
    # _assemble_page). Đồng thời, BẤT KỲ trang nào có ô bảng bị flag (fallback
    # do nghi vietocr hallucination) đều bắt buộc vào diện cần review, kể cả
    # khi confidence trung bình của cả trang vẫn >= 0.85 (1 vài ô sai không đủ
    # kéo trung bình xuống nhưng vẫn cần người kiểm tra lại số liệu đó).
    low_confidence_pages = sorted(
        p for p, c in pages_confidence.items()
        if not (c >= 0.85) or p in pages_flagged_cells
    )

    return {
        "source": "paddleocr_structurev3+vietocr",
        "pages_markdown": pages_markdown,   # text thường + bảng dạng HTML per page
        "pages_confidence": pages_confidence,
        "low_confidence_pages": low_confidence_pages,  # nên review thủ công
        "pages_flagged_cells": pages_flagged_cells,    # (paddle_text vs vietocr_text) từng ô nghi vấn
        "pages_error": pages_error,         # trang nào lỗi + lỗi gì -- KHÔNG được rỗng nếu output toàn bộ trống
        "pages": num_pages,
    }
